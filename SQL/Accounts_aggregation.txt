Phase 2 & Phase 4 : DQL, Jointures, Self-Join, Ensembles & Agrégations
++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++
Requête 2 : Jointures multiples (INNER JOIN / LEFT JOIN) — Enrichir les virements

SELECT 
    t.transaction_id,
    t.tx_timestamp,
    c1.entity_name AS entite_emettrice,
    c2.entity_name AS entite_receptrice,
    t.amount_paid,
    t.payment_currency,
    t.niveau_risque
FROM raw.transactions t
INNER JOIN raw.comptes c1 
    ON t.from_account = c1.account_number AND t.from_bank = c1.bank_id
LEFT JOIN raw.comptes c2 
    ON t.to_account = c2.account_number AND t.to_bank = c2.bank_id
WHERE t.payment_currency IN ('US Dollar', 'Euro')
ORDER BY t.amount_paid DESC
LIMIT 100;

++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++
Requête 3 : Auto-jointure (Self-Join) — Détecter le rebond rapide d'argent (A -> B -> C)

SELECT 
    t1.from_account AS compte_A,
    t1.to_account AS compte_rebond_B,
    t2.to_account AS compte_C,
    t1.amount_paid AS montant_entree_B,
    t2.amount_paid AS montant_sortie_B,
    t1.tx_timestamp AS heure_1,
    t2.tx_timestamp AS heure_2
FROM raw.transactions t1
INNER JOIN raw.transactions t2 
    ON t1.to_account = t2.from_account 
    AND t1.to_bank = t2.from_bank -- Le récepteur du 1er virement devient l'émetteur du 2ème
WHERE t2.tx_timestamp BETWEEN t1.tx_timestamp AND (t1.tx_timestamp + INTERVAL '2 hours')
  AND NOT (t1.from_account = t2.to_account AND t1.from_bank = t2.to_bank)
LIMIT 500;

Requête 4 : Agrégation (GROUP BY + HAVING) — Création de la vue finale pour Python

CREATE VIEW analytics.dataset_ml_aml AS
SELECT 
    t.from_account AS account_number,
    t.from_bank AS bank_id,
    c.bank_name,
    c.entity_name,
    COUNT(t.transaction_id) AS nb_virements_emis,
    COUNT(DISTINCT (t.to_account, t.to_bank)) AS nb_beneficiaires_uniques,
    COUNT(DISTINCT t.to_bank) AS nb_banques_destinatrices,
    SUM(t.amount_paid) AS volume_total_envoye,
    ROUND(AVG(t.amount_paid), 2) AS montant_moyen,
    MAX(t.amount_paid) AS montant_max,
    SUM(CASE WHEN t.payment_format = 'Wire' THEN 1 ELSE 0 END) AS nb_virements_wire,
    SUM(CASE WHEN t.niveau_risque = 'Eleve' THEN 1 ELSE 0 END) AS nb_ops_risque_eleve,
    MAX(t.is_laundering) AS label_blanchiment_reel -- Conservé uniquement pour évaluer le modèle à la fin
FROM raw.transactions t
INNER JOIN raw.comptes c 
    ON t.from_account = c.account_number AND t.from_bank = c.bank_id
WHERE t.payment_currency = 'US Dollar' -- Alternative simple : isole le dollar pour éviter la conversion de devises
GROUP BY t.from_account, t.from_bank, c.bank_name, c.entity_name
HAVING COUNT(t.transaction_id) >= 3; -- Filtre après agrégation : garde uniquement les comptes actifs (3+ virements)
